package main

import (
        "crypto/rsa"
        "crypto/x509"
        "database/sql"
        "encoding/pem"
        "errors"
        "log"
        "net/http"
        "strconv"
        "time"

        "github.com/golang-jwt/jwt/v5"
)

func authHandler(w http.ResponseWriter, r *http.Request) { // /auth
        if r.Method != http.MethodPost {
                http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
                return
        }

        var kid int
        var key []byte
        var exp int64
        var err error
        now := time.Now().Unix()

        //Looks for expired keys if query asked for it, otherwise looks for unexpired keys.
        if r.URL.Query().Has("expired") { //FIXME also like couldn't I just use a bool and half this code
                err = db.QueryRow(
                        "SELECT kid, key, exp FROM keys WHERE exp <= ? LIMIT 1",
                        now,
                ).Scan(&kid, &key, &exp)

        } else {
                err = db.QueryRow(
                        "SELECT kid, key, exp FROM keys WHERE exp > ? LIMIT 1",
                        now,
                ).Scan(&kid, &key, &exp)
        }

        if errors.Is(err, sql.ErrNoRows) { // If no key is found returns an error
                http.Error(w, "No suitable key available", http.StatusInternalServerError)
                return
        } else if err != nil {
                log.Printf("Failed to retrieve key: %v", err)
                http.Error(w, "Failed to retrieve key", http.StatusInternalServerError)
                return
        }

        //deserialize key
        block, _ := pem.Decode(key)
        if block == nil {
                http.Error(w, "Invalid stored key", http.StatusInternalServerError)
                return
        }
        privateKeyImported, err := x509.ParsePKCS8PrivateKey(block.Bytes)
        if err != nil {
                http.Error(w, "Invalid stored key", http.StatusInternalServerError)
                return
        }
        rsaPrivateKey, ok := privateKeyImported.(*rsa.PrivateKey)
        if !ok {
                http.Error(w, "Stored key is not RSA", http.StatusInternalServerError)
                return
        }

        KID := strconv.Itoa(kid)

        //Creates a JWT token with the selected key
        token := jwt.NewWithClaims(
                jwt.SigningMethodRS256,
                jwt.MapClaims{
                        "sub": "test-user",
                        "iat": now,
                        "exp": exp,
                },
        )
        token.Header["kid"] = KID

        //Signs the token with the selected key
        signedToken, err := token.SignedString(rsaPrivateKey)
        if err != nil {
                http.Error(w, "Failed to sign token", http.StatusInternalServerError)
                return
        }

        //sends the signed token to the client
        w.Header().Set("Content-Type", "application/jwt")
        w.Write([]byte(signedToken))
}

import (

        "encoding/pem"
        "log"

)

func jwksHandler(w http.ResponseWriter, r *http.Request) { // /.well-known/jwks.json
        if r.Method != http.MethodGet {
                http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
                return
        }

        jwks := JWKS{
                Keys: []JWK{},
        }

        now := time.Now().Unix()

        //If key is not expired, add it to the list of possible keys to send to the client, converting it to a JWK in the process
        rows, err := db.Query("SELECT kid, key FROM keys WHERE exp > ?", now)
        if err != nil {
                log.Printf("Failed to retrieve key: %v", err)
                http.Error(w, "Failed to retrieve key", http.StatusInternalServerError)
                return
        }
        defer rows.Close()
        for rows.Next() {
                var kid int
                var key []byte
                err = rows.Scan(&kid, &key)
                if err != nil {
                        log.Printf("Failed to retrieve key: %v", err)
                        http.Error(w, "Failed to retrieve key", http.StatusInternalServerError)
                        return
                }

                //deserialize key
                block, _ := pem.Decode(key)
                if block == nil {
                        http.Error(w, "Invalid stored key", http.StatusInternalServerError)
                        return
                }
                privateKeyImported, err := x509.ParsePKCS8PrivateKey(block.Bytes)
                if err != nil {
                        http.Error(w, "Invalid stored key", http.StatusInternalServerError)
                        return
                }
                rsaPrivateKey, ok := privateKeyImported.(*rsa.PrivateKey)
                if !ok {
                        http.Error(w, "Stored key is not RSA", http.StatusInternalServerError)
                        return
                }

                //add to jwks to send
                jwks.Keys = append(jwks.Keys, keyToJWK(rsaPrivateKey, kid))
        }
        if err = rows.Err(); err != nil {
                log.Printf("Failed to retrieve key: %v", err)
                http.Error(w, "Failed to retrieve key", http.StatusInternalServerError)
                return
        }

        //send it
        w.Header().Set("Content-Type", "application/json")
        json.NewEncoder(w).Encode(jwks)
}

package main

import (
        "crypto/rand"
        "crypto/rsa"
        "crypto/x509"
        "encoding/base64"
        "encoding/pem"
        "strconv"
        "time"
)

// Data about the keys that JSON sends to clients
type JWK struct {
        Kty string `json:"kty"`
        N   string `json:"n"`
        E   string `json:"e"`
        Alg string `json:"alg"`
        Use string `json:"use"`
        Kid string `json:"kid"`
}

// Slice of all the acceptable keys (not expired)
type JWKS struct {
        Keys []JWK `json:"keys"`
}

// Use random number to make an RSA key
func generateKey(expiresAt int64) error {
        //make the key
        privateKey, err := rsa.GenerateKey(rand.Reader, 2048)
        if err != nil {
                return err
        }

        //serialize it
        privateKeyBytes, err := x509.MarshalPKCS8PrivateKey(privateKey)
        if err != nil {
                return err
        }
        privatePEM := pem.EncodeToMemory(&pem.Block{
                Type:  "PRIVATE KEY",
                Bytes: privateKeyBytes,
        })

        //add to the db
        _, err = db.Exec("INSERT INTO keys(key, exp) VALUES(?, ?)", privatePEM, expiresAt)
        if err != nil {
                return err
        }

        return nil
}

// used for testing purposes to override the key generation function
var generateKeyFunc = generateKey

// Make a unexpired key and an expired key for proof of concept
func initializeKeys() error {
        now := time.Now()

        //unexpired key has expiry set one hour from starting up the server
        err := generateKeyFunc(now.Add(time.Hour).Unix())
        if err != nil {
                return err
        }

        //expired key has expiry set one hour before starting up the server
        err = generateKeyFunc(now.Add(-time.Hour).Unix())
        if err != nil {
                return err
        }

        return nil
}

// Takes an RSA key and gathers the correct information to make a JWK to send to the client
func keyToJWK(PrivateKey *rsa.PrivateKey, kid int) JWK {
        //convert kid to string for JWK
        KID := strconv.Itoa(kid)

        n := base64.RawURLEncoding.EncodeToString(PrivateKey.N.Bytes())

        eBytes := []byte{
                byte(PrivateKey.E >> 16),
                byte(PrivateKey.E >> 8),
                byte(PrivateKey.E),
        }

        return JWK{
                Kty: "RSA",
                N:   n,
                E:   base64.RawURLEncoding.EncodeToString(eBytes),
                Alg: "RS256",
                Use: "sig",
                Kid: KID,
        }
}

package main

import (
        "database/sql"
        "fmt"
        "net/http"

        _ "github.com/mattn/go-sqlite3"
)

var db *sql.DB //declare db here so it's global
var dbPath = "./totally_not_my_privateKeys.db"
var serverMux *http.ServeMux

// sets up the keys and handlers without actually starting the server
func setupServer() error {
        //Create/open DB file
        var err error
        db, err = sql.Open("sqlite3", dbPath)
        if err != nil {
                return err
        }

        sqlStmt := `
    CREATE TABLE IF NOT EXISTS keys(
            kid INTEGER PRIMARY KEY AUTOINCREMENT,
            key BLOB NOT NULL,
            exp INTEGER NOT NULL
        );
    `
        _, err = db.Exec(sqlStmt)
        if err != nil {
                return err
        }

        //Make keys
        err = initializeKeys()
        if err != nil {
                return err
        }

        serverMux = http.NewServeMux()
        serverMux.HandleFunc("/.well-known/jwks.json", jwksHandler)
        serverMux.HandleFunc("/auth", authHandler)

        return nil
}

func main() {
        err := setupServer()
        if err != nil {
                fmt.Println(err)
                return
        }

        fmt.Println("Server running on http://localhost:8080")

        err = http.ListenAndServe(":8080", serverMux)
        if err != nil {
                fmt.Println(err)
        }
        db.Close()
}
